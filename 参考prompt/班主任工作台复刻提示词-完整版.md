## 二、技术约束（10 条铁律，违反即返工）

1. **单文件**：所有样式与脚本写在同一个 `index.html` 内；禁止拆分文件、禁止 `<script src>` 外链、禁止 `<link>` 外链样式。
2. **零依赖**：不用 React/Vue/jQuery，不用图标字体库，不用外部 API（唯一例外：批量导入 Excel 时首次联网加载 xlsx 解析组件，见 10.4）。
3. **图标**：用内联 SVG（`stroke="currentColor"`、`stroke-width="1.8"`、`viewBox="0 0 24 24"`、线性风格）或 emoji；全部预置进 `ICONS` 对象。
4. **字体**：`"PingFang SC","Microsoft YaHei","Hiragino Sans GB","Noto Sans SC",system-ui,sans-serif`。
5. **兼容性**：Chrome/Edge 最新版即可；移动端（≤860px）响应式适配（侧边栏变抽屉、卡片单列）；仪表盘网格 ≤900px 强制单列。
6. **安全**：所有插入 DOM 的动态文本必须经过 `esc()`（HTML 转义）；删除操作用二次确认（`confirm2`）；文件导出统一加 BOM 防 Excel 乱码。
7. **数据**：对象结构见「五、数据结构」，字段名必须完全一致；`migrate()` 负责旧数据补齐与版本升级；`setting.appVersion` 为应用级、**强制跟随最新版本号，不被旧数据覆盖**。
8. **性能**：渲染函数直接拼接 HTML 字符串赋给 `innerHTML`，无需框架、无需 diff；事件绑定用内联 `onclick="App.xxx()"`（事件处理函数统一挂到 `App` 对象）。
9. **风格**：代码用 ES5 语法（`var`、`function`、`Array.prototype.forEach` 等），避免箭头函数/模板字符串，保证任何 Agent 都能原样重建；缩进 2 空格。
10. **完整性**：27 个导航模块 + 仪表盘 9 区块 + 6 套主题 + 演示数据，**一个都不能少**。

### 4.2 渲染调度（核心，必须照此实现）

```js
function setView(id){
  current = id; mstate = {};
  try{ localStorage.setItem("wbLastView", id); }catch(e){}
  window._scrollToTop = true;
  render();
  if(window.innerWidth <= 860) toggleSidebar(true); // 移动端收起抽屉
}
function render(){
  ensureModuleSwitches();
  var toTop = window._scrollToTop === true; window._scrollToTop = false;
  applyTheme(); setBrandAva(); updateFavicon(); renderTopClass();
  // 品牌标题 = 教师姓名+"的工作台"（姓名末尾已是"老师"则不重复加），同步 document.title
  var bn = DB.setting.teacherName || "老师";
  if(!/老师$/.test(bn)) bn += "老师";
  document.getElementById("brandTitle").textContent = bn + "的工作台";
  document.title = document.getElementById("brandTitle").textContent;
  // 全局班级上下文：始终与仪表盘当前班级一致
  if(DB.classes.length){
    var dc = DB.setting.dashClass || DB.classes[0].id;
    if(!DB.classes.some(function(c){ return c.id === dc; })) dc = DB.classes[0].id;
    filterClass = dc;
  }
  document.getElementById("pageTitle").textContent = TITLES[current] || "仪表盘";
  document.getElementById("pageSub").textContent = current === "dashboard"
    ? new Date().toLocaleDateString("zh-CN",{year:"numeric",month:"long",day:"numeric",weekday:"long"}) : "";
  // 退休徽标（仅日历假期页显示，见 9.16）
  renderNav();
  var v = document.getElementById("view");
  if(current === "dashboard") v.innerHTML = renderDashboard();
  else if(current === "students") v.innerHTML = renderStudents();
  else if(current === "classes") v.innerHTML = renderClasses();
  else if(current === "grades") v.innerHTML = renderGrades();
  else if(current === "attendance") v.innerHTML = renderAttendance();
  else if(current === "homework") v.innerHTML = renderHomework();
  else if(current === "leave") v.innerHTML = renderLeave();
  else if(current === "affairs") v.innerHTML = renderAffairs();
  else if(current === "activities") v.innerHTML = renderActivities();
  else if(current === "home") v.innerHTML = renderHome();
  else if(current === "evaluate") v.innerHTML = renderEvaluate();
  else if(current === "tasks") v.innerHTML = renderTasks();
  else if(current === "safety") v.innerHTML = renderSafety();
  else if(current === "ai") v.innerHTML = renderAI();
  else if(current === "subject") v.innerHTML = renderSubject();
  else if(current === "bookmarks") v.innerHTML = renderBookmarks();
  else if(current === "fun") v.innerHTML = renderFun();
  else if(current === "data") v.innerHTML = renderData();
  if(toTop) v.scrollTop = 0;
  startClock();
}
```

**拖拽排序（含视口边缘自动滚动）**——完整实现：
```js
var _drag = {on:false, fromId:"", overId:"", raf:0, scrollOn:false};
function dashDragStart(e, id){
  if(!mstate.dashEdit) return;
  if(e.cancelable) e.preventDefault();
  _drag = {on:true, fromId:id, overId:id, raf:0, scrollOn:false};
  document.body.style.cursor = "grabbing";
  var el = document.querySelector('.dash-grid>.section[data-mod="'+id+'"]');
  if(el) el.classList.add("dragging");
  window.addEventListener("pointermove", dashDragMove);
  window.addEventListener("pointerup", dashDragEnd);
}
function dashDragMove(e){
  if(!_drag.on) return;
  // 视口边缘自动滚动：顶部/底部 80px 内启动
  var vh = window.innerHeight || document.documentElement.clientHeight;
  if(e.clientY < 80 || e.clientY > vh - 80){
    if(!_drag.scrollOn){ _drag.scrollOn = true; window.requestAnimationFrame(dashAutoScroll); }
  } else { _drag.scrollOn = false; if(_drag.raf){ window.cancelAnimationFrame(_drag.raf); _drag.raf = 0; } }
  dashDragPick(e.clientX, e.clientY);
}
function dashDragPick(x, y){
  var el = document.elementFromPoint(x, y);
  while(el && !(el.classList && el.classList.contains("section") && el.dataset && el.dataset.mod))
    el = el.parentElement;
  if(!el || el.dataset.mod === _drag.fromId){
    var oe = document.querySelector('.dash-grid>.section.drag-over');
    if(oe) oe.classList.remove("drag-over");
    return;
  }
  if(el.dataset.mod !== _drag.overId){
    var old = document.querySelector('.dash-grid>.section.drag-over');
    if(old) old.classList.remove("drag-over");
    _drag.overId = el.dataset.mod;
    el.classList.add("drag-over");
    swapDashOrder(_drag.fromId, _drag.overId);
  }
}
function dashAutoScroll(){
  if(!_drag.on || !_drag.scrollOn) return;
  var se = document.scrollingElement || document.documentElement;
  var y = _drag.lastY;
  var vh = window.innerHeight || document.documentElement.clientHeight;
  var speed = 0;
  if(y < 80)      speed = Math.max(6, Math.round((80 - y) / 8));       // 顶部：向上滚 6~22px/帧
  else if(y > vh - 80) speed = -Math.max(6, Math.round((y - (vh-80)) / 8)); // 底部：向下滚
  if(speed) se.scrollTop += speed;
  _drag.raf = window.requestAnimationFrame(dashAutoScroll);
}
function swapDashOrder(a, b){
  var L = getDashLayout();
  var ia=-1, ib=-1;
  L.forEach(function(x,i){ if(x.id===a) ia=i; if(x.id===b) ib=i; });
  if(ia<0 || ib<0) return;
  var t = L[ia]; L[ia] = L[ib]; L[ib] = t;
  var elA = document.querySelector('.dash-grid>.section[data-mod="'+a+'"]');
  var elB = document.querySelector('.dash-grid>.section[data-mod="'+b+'"]');
  if(elA && elB){ elA.style.order = ib; elB.style.order = ia; }
}
function dashDragEnd(){
  if(!_drag.on) return;
  _drag.on = false; _drag.scrollOn = false;
  if(_drag.raf){ window.cancelAnimationFrame(_drag.raf); _drag.raf = 0; }
  document.body.style.cursor = "";
  document.querySelectorAll(".dash-grid .section.dragging,.dash-grid .section.drag-over")
    .forEach(function(x){ x.classList.remove("dragging","drag-over"); });
  save(); render();
  window.removeEventListener("pointermove", dashDragMove);
  window.removeEventListener("pointerup", dashDragEnd);
}
```

**右缘手柄调宽**——完整实现（★ 基准必须固定，严禁用实时档位参与换算）：
```js
function dashResizeStart(e, id){
  if(!mstate.dashEdit) return;
  e.preventDefault(); e.stopPropagation();
  var el = document.querySelector('.dash-grid>.section[data-mod="'+id+'"]');
  if(!el) return;
  var item = null; getDashLayout().forEach(function(x){ if(x.id===id) item=x; });
  if(!item) return;
  var grid = document.getElementById("dashGrid");
  var startX = e.clientX;
  var cellW = grid.offsetWidth / 12;
  // 用「拖拽开始时的实际渲染宽度」为固定基准（不用 item.col*cellW！否则档位一跳基准漂移）
  var rect = el.getBoundingClientRect();
  var startW = (rect && cellW) ? (rect.left + rect.width - item.col*cellW) : 0; // 修正 gap 偏差
  el.classList.add("resizing");
  document.body.style.cursor = "ew-resize";
  function move(ev){
    var dx = ev.clientX - startX;
    if(!cellW) return;
    var ns = Math.round((startW + dx) / cellW);   // 固定基准 + 位移，去程回程完全对称
    ns = Math.max(4, Math.min(12, ns));
    if(ns % 2) ns += dx >= 0 ? 1 : -1;            // 取偶：4/6/8/10/12
    ns = Math.max(4, Math.min(12, ns));
    if(ns !== item.col){
      item.col = ns;
      el.className = el.className.replace(/col-\d+/, "col-"+ns);
    }
  }
  function up(){
    window.removeEventListener("pointermove", move);
    window.removeEventListener("pointerup", up);
    document.body.style.cursor = "";
    el.classList.remove("resizing");
    save(); render();
  }
  window.addEventListener("pointermove", move);
  window.addEventListener("pointerup", up);
}
```

## 十二、常见坑（避免踩，全部有教训）

1. **拼接 HTML 忘记转义** → 所有动态文本一律过 `esc()`，否则 XSS 且中文引号会被截断。
2. **日历标题与标签混用 `.split("")`** → 会把中文拆成单字导致标签散架；标题用整体字符串拼接。
3. **SVG 拉伸变形** → 固定 `viewBox` 并用 CSS 限制容器比例（`.linechart{max-width:400px;height:150px}`）。
4. **弹窗默认 560px 装不下宽表格** → 成绩录入/导入预览等宽内容用 `modal-wide` 类（96vw/1200px）。
5. **农历数据表不完整** → lunarInfo 必须完整 201 项（1900-2100），缺项闰月/年份错乱。
6. **localStorage 被旧版本数据污染** → `migrate()` 必须补齐所有缺失字段；`appVersion` 强制覆盖。
7. **移动端横向溢出** → 表格外层套 `.scroll-x`，网格用 auto-fill minmax。
8. **调宽换算用实时档位做基准** → 档位一跳基准漂移，1/2 宽档成死区回不去；**必须以拖拽开始时的实际渲染宽度为固定基准**。
9. **拖拽排序不处理视口边缘** → 目标模块跑出可视区永远命中不了；必须实现边缘自动滚动 + 每帧重算目标。
10. **座位三人桌出现「虚假列」** → 列数不可被桌人数整除时，只渲染实际存在的座位，禁止渲染/写入超界列。
11. **导出 CSV 无 BOM** → Excel 打开中文乱码；必须 `"\uFEFF"` 前缀。
12. **删除班级/学生级联混乱** → 删除操作统一二次确认；被引用数据保留不级联删除（避免脏数据）。
13. **弹窗内选择按钮重建整个弹窗**（日历添加待办点紧急程度/四象限、节假日选图标）→ 会清空用户已输入内容；只切换选中样式（`data-val` + 局部 className 更新），禁止重建。
14. **`file://` 空 href 链接点击** → 触发 `Unsafe attempt to load URL` 报错；动态渲染链接前校验 URL 合法（`/^https?:\/\//i`），空/非法地址渲染为提示块；页面加全局空 href 拦截。
15. **页面渲染无异常兜底** → 旧数据不兼容时白屏（普通模式打不开、无痕正常）；渲染分发包 try-catch，失败显示「一键修复/重置数据」入口，另备 `fixAndReload()` 备份损坏数据后重建。
